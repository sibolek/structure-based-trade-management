    private static MarketDepthEventArgs Depth(SuperDom dom,MarketDataType side,Operation operation,int pos,double price,long qty)
    {
        List<MarketDepthRow> book=side==MarketDataType.Ask?dom.MarketDepth.Asks:dom.MarketDepth.Bids;
        if(operation==Operation.Add)book.Insert(pos,new MarketDepthRow{Price=price,Volume=qty});
        else if(operation==Operation.Update)book[pos]=new MarketDepthRow{Price=price,Volume=qty};
        else book.RemoveAt(pos);
        PriceRow row=dom.Rows.FirstOrDefault(delegate(PriceRow r){return r.Price==price;});
        if(row==null){row=new PriceRow{Price=price};dom.Rows.Add(row);}
        if(side==MarketDataType.Ask)row.AskVolume=operation==Operation.Remove?0:qty;
        else row.BidVolume=operation==Operation.Remove?0:qty;
        return new MarketDepthEventArgs{Instrument=dom.Instrument,MarketDataType=side,Operation=operation,Position=pos,Price=price,Volume=qty,Time=Connection.PlaybackConnection.Now};
    }
    private static MarketDataEventArgs Last(SuperDom dom,double price,long volume,DateTime source,double bid=100,double ask=100.25)
    {return new MarketDataEventArgs{Instrument=dom.Instrument,MarketDataType=MarketDataType.Last,Price=price,Volume=volume,Bid=bid,Ask=ask,Time=source};}
