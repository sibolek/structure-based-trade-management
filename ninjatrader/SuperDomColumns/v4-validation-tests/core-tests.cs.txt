using System;
using System.Collections.Generic;
using System.Diagnostics;

using R = ExecutionOSRollingLiquidityV4;
internal static class Program
{
    private static int passed;
    private static readonly List<string> results = new List<string>();
    private static long T(double seconds) { return (long)(seconds * TimeSpan.TicksPerSecond); }
    private static long Source(double seconds) { return new DateTime(2026, 10, 12).Ticks + T(seconds); }
    private static R New(int seconds = 5, int grace = 250, int depth = 10, int cap = 200000)
    { R r = new R(seconds, grace, depth, cap); r.Reset(0); return r; }
    private static void D(R r, R.Side side, R.DepthAction action, int pos, long price, long qty, double source, double receipt)
    { r.Depth(side, action, pos, price, qty, Source(source), T(receipt)); }
    private static void Seed(R r) { D(r, R.Side.Ask, R.DepthAction.Add, 0, 400, 100, 0, 0); }
    private static long V(R r, R.Side side = R.Side.Ask, long price = 400)
    { long v; if (!r.TryValue(side, price, out v)) throw new Exception("Value unavailable"); return v; }
    private static void Eq(long actual, long expected, string why)
    { if (actual != expected) throw new Exception(why + ": expected " + expected + ", got " + actual); }
    private static void Yes(bool actual, string why) { if (!actual) throw new Exception(why); }
    private static void Test(string name, Action test)
    { test(); passed++; results.Add("PASS " + name); Console.WriteLine("PASS " + name); }
    public static int Main()

        try
        {

                R r = New(); Seed(r); Eq(V(r), 0, "initial");

                Eq(V(r, R.Side.Bid, 399), 0, "bootstrap");

                D(r, R.Side.Ask, R.DepthAction.Update, 0, 400, 90, 2, 2); Eq(V(r), -10, "net");
            });
            Test("Pure execution trade-first", delegate {
                R r = New(); Seed(r); r.Trade(R.Side.Ask,400,30,Source(1),T(1));
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,70,1.01,1.01);
                Eq(V(r),0,"execution"); Eq(r.MatchedVolume,30,"credit"); Eq(r.PendingCount,0,"consumed");
            });
            Test("Pure execution depth-first using event timestamps", delegate {
                R r = New(); Seed(r); D(r,R.Side.Ask,R.DepthAction.Update,0,400,70,1.01,1);
                r.Trade(R.Side.Ask,400,30,Source(1),T(1.05)); Eq(V(r),0,"execution");

            Test("Execution plus additions and unchanged-depth replenishment", delegate {
                R r=New(); Seed(r); r.Trade(R.Side.Ask,400,30,Source(1),T(1));
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,120,1.01,1.01); Eq(V(r),50,"replenishment");
                r.Trade(R.Side.Ask,400,20,Source(2),T(2));
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,120,2.01,2.01); Eq(V(r),70,"unchanged depth");
            });
            Test("Execution credit consumed once across subsequent depth updates", delegate {
                R r=New(); Seed(r); r.Trade(R.Side.Ask,400,30,Source(1),T(1));
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,70,1.01,1.01);
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,60,1.02,1.02);
                Eq(V(r),-10,"no duplicate"); Eq(r.MatchedVolume,30,"one credit");

            Test("Split executions, coalesced depth", delegate {
                R r=New(); Seed(r); r.Trade(R.Side.Ask,400,10,Source(1),T(1));
                r.Trade(R.Side.Ask,400,20,Source(1.01),T(1.01));

            });
            Test("Independent bid/ask at same price and independent price levels", delegate {
                R r=New(); Seed(r); D(r,R.Side.Bid,R.DepthAction.Add,0,400,100,0,0);
                D(r,R.Side.Ask,R.DepthAction.Add,1,401,100,0,0);
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,70,1.01,1);

                R r=New(); Seed(r); D(r,R.Side.Ask,R.DepthAction.Update,0,400,140,1,1);

                r.Trade(R.Side.Ask,400,30,Source(1),T(2)); Eq(V(r),0,"expired remains zero"); Eq(r.MatchedVolume,0,"no expired credit");

            Test("Unconfirmed executions expire without invented stacking", delegate {
                R r=New(); Seed(r); r.Trade(R.Side.Ask,400,30,Source(1),T(1));
                r.Advance(T(1.26)); Eq(V(r),0,"no confirmation"); Eq(r.UnmatchedVolume,30,"diagnostic");
                D(r,R.Side.Ask,R.DepthAction.Update,0,400,70,1.4,1.4); Eq(V(r),-30,"too late to credit");

            Test("Contradictory event timestamps stay unmatched", delegate {
                R r=New(); Seed(r); D(r,R.Side.Ask,R.DepthAction.Update,0,400,70,1,1);
                r.Trade(R.Side.Ask,400,30,Source(1.1),T(1.02)); Eq(V(r),-30,"no backward pairing");

            Test("Missing source timestamp blocks execution attribution", delegate {
                R r=New(); r.Depth(R.Side.Ask,R.DepthAction.Add,0,400,100,0,0);

            Test("Duplicate price rows aggregate side quantity", delegate {

    private static void Benchmark(int seconds)
    {
        R r=New(seconds,250,10,500000); Seed(r); const int events=1000000;
        long allocatedBefore=GC.GetTotalAllocatedBytes(true); Stopwatch timer=Stopwatch.StartNew();
        for(int i=1;i<=events;i++){
            long now=i*1000L; // Synthetic 10,000 depth updates per market second.
            r.Depth(R.Side.Ask,R.DepthAction.Update,0,400,100+(i%10),Source(0)+now,now);
        }
        timer.Stop(); Yes(!r.Faulted,"benchmark capacity");
        double ms=timer.Elapsed.TotalMilliseconds;
        long allocated=GC.GetTotalAllocatedBytes(true)-allocatedBefore;
        string line=String.Format(CultureInfo.InvariantCulture,
            "BENCH horizon={0}s events={1} wallMs={2:F1} eventsPerSecond={3:F0} allocatedBytes={4} retainedEntries={5}",
            seconds,events,ms,events*1000.0/ms,allocated,r.EntryCount);
        results.Add(line);Console.WriteLine(line);
        r.Advance(events*1000L+T(seconds));Eq(r.EntryCount,0,"quiet expiry benchmark");
    }
    private static void MultiLevelBenchmark()
    {
        R r=New(30,250,10,500000);
        for(int i=0;i<10;i++){
            r.Depth(R.Side.Bid,R.DepthAction.Add,i,399-i,100,Source(0),0);
            r.Depth(R.Side.Ask,R.DepthAction.Add,i,400+i,100,Source(0),0);
        }
        const int events=1000000;Stopwatch timer=Stopwatch.StartNew();long before=GC.GetTotalAllocatedBytes(true);
        for(int i=1;i<=events;i++){
            long now=i*1000L;int pos=(i/2)%10;R.Side side=i%2==0?R.Side.Bid:R.Side.Ask;
            long price=side==R.Side.Bid?399-pos:400+pos;
            if(i%10==0)r.Trade(side,price,1,Source(0)+now,now);
            r.Depth(side,R.DepthAction.Update,pos,price,100+(i%7),Source(0)+now+10,now+10);
        }
        timer.Stop();Yes(!r.Faulted,"multi-level capacity");Eq(r.MatchedVolume,100000,"multi-level matched execution count");
        string line=String.Format(CultureInfo.InvariantCulture,
            "BENCH multiLevel=20 horizon=30s events={0} wallMs={1:F1} eventsPerSecond={2:F0} allocatedBytes={3} retainedEntries={4} matchedVolume={5}",
            events,timer.Elapsed.TotalMilliseconds,events*1000.0/timer.Elapsed.TotalMilliseconds,GC.GetTotalAllocatedBytes(true)-before,r.EntryCount,r.MatchedVolume);
        results.Add(line);Console.WriteLine(line);
    }
