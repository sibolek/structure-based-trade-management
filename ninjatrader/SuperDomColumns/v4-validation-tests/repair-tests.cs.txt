 class OTrade{internal long Id,Time,Price,Volume,Matched;internal int Direction;}

 static object Get(object o,string n){return o.GetType().GetField(n,BindingFlags.NonPublic|BindingFlags.Instance).GetValue(o);}
 static void Oracle(){
  F f=New();Random r=new Random(68721);List<OTrade> trades=new List<OTrade>();List<ORefill> refills=new List<ORefill>();long serial=0,now=T(5);
  for(int i=0;i<8000;i++){
   now+=r.Next(1,7)*TimeSpan.TicksPerMillisecond;int op=r.Next(4);
   if(op==0||trades.Count==0){int d=r.Next(3)-1;long price=99+r.Next(4),vol=1+r.Next(70);long id=f.AddTrade(d,price,vol,100,101,now);trades.Add(new OTrade{Id=id,Time=now,Price=price,Volume=vol,Direction=d});}
   else if(op==1){OTrade t=trades[r.Next(trades.Count)];long vol=r.Next(80);f.Advance(now);f.MatchTrade(t.Id,vol);if(t.Time>now-T(5))t.Matched=Math.Min(t.Volume,t.Matched+vol);}
   else if(op==2){F.Side side=r.Next(2)==0?F.Side.Bid:F.Side.Ask;long price=side==F.Side.Bid?100:101,net=r.Next(-100,101);ORefill x=new ORefill{Id=++serial,Time=now,Price=price,Net=net,Side=side};refills.Add(x);f.Contribution(x.Id,side,price,net,now,now);}
   else{ORefill x=refills.Count==0?null:refills[r.Next(refills.Count)];if(x!=null){long net=r.Next(-100,101);f.Contribution(x.Id,x.Side,x.Price,net,x.Time,now);if(x.Time>now-T(5.25))x.Net=net;}}
   f.Evaluate(now);
   foreach(bool buy in new[]{true,false}){
    int direction=buy?1:-1;bool started=false,breached=false;long peak=0;
    foreach(OTrade t in trades){if(t.Time<=now-T(5))continue;if(!started){if(t.Direction!=direction)continue;started=true;peak=t.Price;}
     if(buy? t.Price>peak:t.Price<peak)peak=t.Price;if((buy?peak-t.Price:t.Price-peak)>1)breached=true;
    }
    object m=Get(f,buy?"buys":"sells");bool actual=(bool)m.GetType().GetProperty("PathBreached",BindingFlags.NonPublic|BindingFlags.Instance).GetValue(m,null);Check(actual==breached,"independent retained path step"+i+" buy="+buy);
   }
   IDictionary levels=(IDictionary)Get(f,"levels");foreach(DictionaryEntry entry in levels){object key=entry.Key,level=entry.Value;F.Side side=(F.Side)Get(key,"Side");long price=(long)Get(key,"Price");object m=Get(level,"Metrics");int d=side==F.Side.Bid?-1:1;
    long executed=0,eligible=0,matched=0,opposite=0,first=long.MinValue,last=long.MinValue,adverse=0,net=0,count=0;
    foreach(OTrade t in trades){if(t.Time<=now-T(5))continue;if(t.Direction==d&&t.Price==price){executed+=t.Volume;if(t.Matched==t.Volume||now-t.Time>T(.25)){eligible+=t.Volume;matched+=t.Matched;}if(first==long.MinValue)first=t.Time;last=t.Time;}else if(t.Direction!=0&&t.Direction!=d&&Math.Abs(t.Price-price)<=1)opposite+=t.Volume;if(first!=long.MinValue)adverse=Math.Max(adverse,side==F.Side.Bid?price-t.Price:t.Price-price);}
    foreach(ORefill x in refills)if(first!=long.MinValue&&x.Side==side&&x.Price==price&&x.Time>now-T(5)&&x.Time>=first-T(.25)){net+=x.Net;if(x.Net>0)count++;}
    Eq((long)Get(m,"Executed"),executed,"executed step"+i);Eq((long)Get(m,"Eligible"),eligible,"eligible step"+i);Eq((long)Get(m,"Matched"),matched,"matched step"+i);Eq((long)Get(m,"Opposite"),opposite,"opposite step"+i);Eq((long)Get(m,"First"),first,"first step"+i);Eq((long)Get(m,"Last"),last,"last step"+i);Eq((long)Get(m,"Adverse"),adverse,"adverse step"+i);Eq((long)Get(m,"Replenished"),net,"net step"+i);Eq((int)Get(m,"Refills"),count,"refills step"+i);
   }
